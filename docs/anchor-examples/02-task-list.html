<form @TaskForm>
  <input @TaskInput type="text" placeholder="New task">
  <input type="submit" value="Add task">
</form>

<ol>
  <repeat as="t" in="$Tasks">
    <li>[t.title] <small>(#[loop.number] of [loop.length])</small></li>
    <else><li class="empty">No tasks yet.</li></else>
  </repeat>
</ol>

<call fn="addTask" params="@TaskInput" await="TaskForm:submit"></call>

<script>
  let { Tasks } = Anchor.expose({ Tasks: [{ title: "Read the Anchor documentation" }] });

  function addTask(title) {
    if (!title) return;
    Tasks.push({ title });
    Anchor.elements.TaskInput.value = "";
  }
  Anchor.exposeFunctions({ addTask });
</script>

Demonstrates: Anchor.expose, <repeat> with the loop object and an empty-state <else>, <call> with Anchor.exposeFunctions.
