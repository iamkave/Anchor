<script src="https://cdn.jsdelivr.net/gh/iamkave/Anchor@1.0.0/interpreter.js"></script>

<label>Name <input type="text" bind="$user.name"></label>
<label>Age <input type="number" bind="$user.age"></label>
<label><input type="checkbox" bind="$user.subscribed"> Subscribe to updates</label>

<div class="out">
  <if test="$user.age" gte="18">Adult account</if>
  <else>Minor account</else>
  <br>
  <if test="$user.subscribed" eq="true">Newsletter: on</if>
  <else>Newsletter: off</else>
</div>

<h3>Current data</h3>
<dl>
  <repeat as="key, value" in="$user">
    <dt>[key]</dt><dd>[value]</dd>
  </repeat>
</dl>

<script>
  Anchor.expose({ user: { name: "Ada", age: 20, subscribed: false } });
</script>

Demonstrates: bind on text, number, and checkbox inputs, <if> with comparators, <repeat> over an object.
