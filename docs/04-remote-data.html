Remote data

Users are loaded from a public API. The list can be sorted, resized, and shows loading and error states.

<script src="https://cdn.jsdelivr.net/gh/iamkave/Anchor@1.0.0/interpreter.js"></script>

<fetch @LoadUsers
       url="https://jsonplaceholder.typicode.com/users"
       pushto="$users"></fetch>

<if test="$usersLoading"><p class="status">Loading</p></if>
<if test="$usersError" has><p class="status">Unable to load users.</p></if>

<div class="row">
  <button @Fewer type="button">Show fewer</button>
  <button @More type="button">Show more</button>
</div>

<ul>
  <repeat as="u" in="$users" sort="name" limit="$pageSize">
    <li><b>[u.name]</b><br><small>[u.email], [u.company.name]</small></li>
    <else><li class="empty">No users to show.</li></else>
  </repeat>
</ul>

<call fn="fewer" await="Fewer:click"></call>
<call fn="more" await="More:click"></call>

<script>
  Anchor.expose({ users: [], pageSize: 3 });

  Anchor.exposeFunctions({
    fewer: () => Anchor.set("pageSize", n => n - 1),
    more:  () => Anchor.set("pageSize", n => n + 1)
  });
</script>

Demonstrates: <fetch> with automatic Loading and Error values, sort, limit, Anchor.set with an updater function.
